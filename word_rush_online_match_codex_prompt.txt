6. Friend Match: Create Room

function generateRoomCode() {
  return Math.random().toString(36).slice(2, 8).toUpperCase();
}

function ensureSupabaseClient() {
  if (supabaseClient) return supabaseClient;
  return initializeSupabaseClient();
}

async function createFriendRoom() {
  const client = ensureSupabaseClient();
  if (!client) return;

  const playerId = getOrCreatePlayerId();
  const roomCode = generateRoomCode();

  const room = {
    room_code: roomCode,
    mode: "friend",
    status: "waiting",
    host_id: playerId,
    host_name: "Host",
    scores: { A: 0, B: 0 },
    current_round: 0
  };

  const { data, error } = await client
    .from("word_rush_rooms")
    .insert(room)
    .select()
    .single();

  if (error) {
    console.error(error);
    setFriendMatchStatus("Failed to create room.");
    return;
  }

  onlineState.enabled = true;
  onlineState.mode = "friend";
  onlineState.roomId = data.id;
  onlineState.roomCode = data.room_code;
  onlineState.playerId = playerId;
  onlineState.playerRole = "host";
  onlineState.isHost = true;

  showWaitingRoom(data);
  await subscribeRoom(data.id);
}

7. Friend Match: Join Room

async function joinFriendRoom(roomCode) {
  const client = ensureSupabaseClient();
  if (!client) return;

  const playerId = getOrCreatePlayerId();
  const code = String(roomCode || "").trim().toUpperCase();

  if (!code) {
    setFriendMatchStatus("Room code is required.");
    return;
  }

  const { data: room, error: findError } = await client
    .from("word_rush_rooms")
    .select("*")
    .eq("room_code", code)
    .eq("status", "waiting")
    .single();

  if (findError || !room) {
    console.error(findError);
    setFriendMatchStatus("Room not found or already started.");
    return;
  }

  const { data: updatedRoom, error: updateError } = await client
    .from("word_rush_rooms")
    .update({
      guest_id: playerId,
      guest_name: "Guest",
      updated_at: new Date().toISOString()
    })
    .eq("id", room.id)
    .is("guest_id", null)
    .select()
    .single();

  if (updateError) {
    console.error(updateError);
    setFriendMatchStatus("Failed to join room.");
    return;
  }

  onlineState.enabled = true;
  onlineState.mode = "friend";
  onlineState.roomId = updatedRoom.id;
  onlineState.roomCode = updatedRoom.room_code;
  onlineState.playerId = playerId;
  onlineState.playerRole = "guest";
  onlineState.isHost = false;

  showWaitingRoom(updatedRoom);
  await subscribeRoom(updatedRoom.id);
}

async function subscribeRoom(roomId) {
  const client = ensureSupabaseClient();
  if (!client) return;

  if (onlineState.channel) {
    await client.removeChannel(onlineState.channel);
    onlineState.channel = null;
  }

  const channel = client
    .channel(`word_rush_room_${roomId}`)
    .on(
      "postgres_changes",
      {
        event: "*",
        schema: "public",
        table: "word_rush_rooms",
        filter: `id=eq.${roomId}`
      },
      (payload) => {
        handleRoomUpdate(payload.new);
      }
    )
    .on(
      "postgres_changes",
      {
        event: "INSERT",
        schema: "public",
        table: "word_rush_room_events",
        filter: `room_id=eq.${roomId}`
      },
      (payload) => {
        handleRoomEvent(payload.new);
      }
    )
    .subscribe();

  onlineState.channel = channel;
}

handleRoomUpdate(room) では以下を行ってください。

- waiting room 表示を更新
- guest_id が入ったら Host側だけ countdown開始
- status が countdown / playing / finished になったらUI反映
- question_set が入ったら onlineState.questionSet に保存

Host側だけが開始処理を行うようにしてください。

if (onlineState.isHost && room.status === "waiting" && room.guest_id) {
  await startOnlineCountdownAsHost(room);
}

ただし、同じ更新を複数回受けて countdown が重複しないように、onlineState.countdownStarted を使ってください。

9. Hostが question_set を生成する
============================================================

オンライン対戦では、両端末で別々に問題生成するとズレる可能性があります。
そのため、Hostだけが試合開始前に question_set を生成してください。

function buildOnlineQuestionSet(count = 20) {
  const questions = [];
  const usedKeys = new Set();

  for (let i = 0; i < count; i++) {
    const q = buildRandomQuestionFromDictionary();
    if (!q) break;

    const key = getQuestionKey(q);
    if (usedKeys.has(key)) continue;
    usedKeys.add(key);

    const cleanQuestion = {
      round: questions.length + 1,
      type: q.type,
      start: q.start || null,
      end: q.end || null,
      side: q.side || null,
      lengthMode: q.lengthMode || null,
      minLen: q.minLen || null,
      maxLen: q.maxLen || null,
      direction: q.direction || null,
      n: q.n || null,
      letter: q.letter || null,
      letters: q.letters || null,
      sequence: q.sequence || null,
      candidateCount: q.candidateCount || null,
      label: getQuestionLabel(q)
    };

    questions.push(cleanQuestion);
  }

  return questions;
}

注意：
- candidates 全リストは保存しないでください。大きすぎます。
- 保存するのは問題条件だけです。
- Guest側はこの question_set を使って同じ問題を表示します。

10. Countdown開始
============================================================

Host側で2人揃ったら：

1. question_set を生成
2. room.status = "countdown"
3. room.question_set = questionSet
4. countdown_started event をinsert
5. 3秒後に room.status = "playing"
6. startOnlineMatch()

async function startOnlineCountdownAsHost(room) {
  if (onlineState.countdownStarted) return;
  onlineState.countdownStarted = true;

  const client = ensureSupabaseClient();
  const questionSet = buildOnlineQuestionSet(20);

  await client
    .from("word_rush_rooms")
    .update({
      status: "countdown",
      question_set: questionSet,
      updated_at: new Date().toISOString()
    })
    .eq("id", room.id);

  await publishRoomEvent("countdown_started", {
    startsInMs: 3000
  });

  setWaitingRoomStatus("Starting in 3...");

  setTimeout(async () => {
    await client
      .from("word_rush_rooms")
      .update({
        status: "playing",
        started_at: new Date().toISOString(),
        updated_at: new Date().toISOString()
      })
      .eq("id", room.id);

    await publishRoomEvent("match_started", {});
    startOnlineMatch();
  }, 3000);
}

Guest側は countdown_started event を受けたら waiting room に Starting... を表示し、match_started を受けたら startOnlineMatch() してください。

13. publishRoomEvent
============================================================

async function publishRoomEvent(eventType, payload = {}) {
  const client = ensureSupabaseClient();
  if (!client || !onlineState.roomId) return;

  const { error } = await client.from("word_rush_room_events").insert({
    room_id: onlineState.roomId,
    event_type: eventType,
    player_id: onlineState.playerId,
    player_role: onlineState.playerRole,
    payload
  });

  if (error) {
    console.error("Failed to publish event:", eventType, error);
  }
}

14. handleRoomEvent
============================================================

function handleRoomEvent(event) {
  const payload = event.payload || {};
  const isMine = event.player_id === onlineState.playerId;

  if (event.event_type === "countdown_started") {
    setWaitingRoomStatus("Starting...");
    return;
  }

  if (event.event_type === "match_started") {
    startOnlineMatch();
    return;
  }

  if (event.event_type === "word_accepted") {
    if (isMine) return;

    const localPlayer = "B";
    const word = payload.word;
    const points = payload.points || getEffectiveWordScore(word);

    state.roundWords[localPlayer].push(word);
    state.scores[localPlayer] += points;
    state.roundRoundScores[localPlayer] += points;

    updateUI();
    return;
  }

  if (event.event_type === "match_finished") {
    finishOnlineMatch(payload.winner || null);
    return;
  }
}

16. Cancel
============================================================

Waiting中にCancelを押したら：

- room status を abandoned にする
- random queue を cancelled にする
- channel解除
- onlineState初期化
- startScreenに戻る

async function cancelOnlineMatch() {
  const client = ensureSupabaseClient();

  if (client && onlineState.roomId) {
    await client
      .from("word_rush_rooms")
      .update({
        status: "abandoned",
        updated_at: new Date().toISOString()
      })
      .eq("id", onlineState.roomId);
  }

  if (client && onlineState.queueId) {
    await client
      .from("word_rush_random_queue")
      .update({
        status: "cancelled",
        updated_at: new Date().toISOString()
      })
      .eq("id", onlineState.queueId);
  }

  if (client && onlineState.channel) {
    await client.removeChannel(onlineState.channel);
  }

  resetOnlineState();
  showStartScreen();
}
